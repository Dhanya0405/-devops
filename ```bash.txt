```bash
#!/bin/bash

echo "================================"
echo "     LINUX SERVER MONITORING"
echo "================================"

# Current Date and Time
echo ""
echo "Date and Time:"
date

# System Uptime
echo ""
echo "System Uptime:"
uptime

# CPU Usage
echo ""
echo "CPU Usage:"
top -bn1 | grep "Cpu(s)"

# Memory Usage
echo ""
echo "Memory Usage:"
free -h

# Disk Usage
echo ""
echo "Disk Usage:"
df -h

# Logged-in Users
echo ""
echo "Logged-in Users:"
who

# Top 5 Running Processes
echo ""
echo "Top 5 Running Processes:"
ps -eo pid,comm,%cpu,%mem --sort=-%cpu | head -n 6

echo ""
echo "================================"
echo "Monitoring Completed!"
echo "================================"
```
